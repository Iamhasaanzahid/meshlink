STEP 2: Build & Upload APK
```bash
cd /home/mhzaly/Documents/Default\ Project/meshlink_android

# Build the release APK (installable directly on phones)
# signingConfig in build.gradle is conditional on KEYSTORE_FILE env var.
# Without it, AGP builds an unsigned release APK.
# To build signed: export KEYSTORE_FILE=/path/to/release.keystore
#                 export KEYSTORE_PASSWORD=your_password
#                 export KEY_ALIAS=meshlink
#                 export KEY_PASSWORD=your_key_password

# Build APK (installable directly on phones, not AAB)
./gradlew assembleRelease \
  -PmeshlinkApiUrl="https://meshlink-api.onrender.com" \
  --no-daemon

# APK location:
# app/build/outputs/apk/release/app-release.apk (signed) or app-release-unsigned.apk (unsigned)

# Upload to GitHub Release
gh release upload v1.0.0 \
  app/build/outputs/apk/release/app-release.apk \
  --repo Iamhasaanzahid/meshlink
```
